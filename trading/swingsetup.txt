<script>
(function(){
  const STORAGE_KEY = "preflight_checklist_v1";

  const sections = [
    { id:"trend", title:"Trend Analysis", items:[
      "Daily trend identified",
      "Weekly trend agrees",
      "No conflicting higher-timeframe structure"
    ]},
    { id:"volume", title:"Volume Profile", items:[
      "Price at HVN / LVN / VAH / VAL",
      "Volume profile shape confirms bias",
      "No unfilled volume gap nearby"
    ]},
    { id:"vwap", title:"VWAP", items:[
      "Price above/below VWAP, matching bias",
      "VWAP slope agrees with trend",
      "No rejection off VWAP band against bias"
    ]},
    { id:"entry", title:"Entry Confirmation", items:[
      "Pullback complete",
      "Strong confirmation candle",
      "Volume confirms the move"
    ]},
    { id:"risk", title:"Risk Management", items:[
      "Risk:Reward is 1:2 or better",
      "Stop placed at technical invalidation",
      "Position size calculated",
      "Risk within 0.5%–1% of account"
    ]},
    { id:"trademgmt", title:"Trade Management", items:[
      "No major news due soon",
      "TP levels / partials planned before entry",
      "Plan set for managing the trade, not reacting"
    ]},
    { id:"personal", title:"Personal Rules", items:[
      "Setup matches my plan exactly",
      "Emotion level: calm",
      "Would take this trade on a demo account",
      "Not trading to recover a prior loss",
      "Not trading out of boredom or FOMO"
    ]}
  ];

  const psychItems = [
    "Am I trading because I'm bored?",
    "Am I trying to recover a loss?",
    "Am I afraid of missing out?",
    "Would I take this trade on a demo?",
    "Does this exactly match my plan?"
  ];

  const sectionsEl = document.getElementById("sections");
  const psychEl = document.getElementById("psychItems");

  // Build main sections
  sections.forEach((sec, sIdx) => {
    const card = document.createElement("div");
    card.className = "card";
    card.innerHTML = `
      <div class="card-head">
        <span class="card-num">${String(sIdx+1).padStart(2,"0")}</span>
        <span class="card-title">${sec.title}</span>
        <span class="card-count" data-count="${sec.id}">0/${sec.items.length}</span>
      </div>
    `;
    sec.items.forEach((text, iIdx) => {
      const id = `${sec.id}_${iIdx}`;
      const row = document.createElement("div");
      row.className = "item";
      row.innerHTML = `<input type="checkbox" id="${id}"><label for="${id}">${text}</label>`;
      card.appendChild(row);
    });
    sectionsEl.appendChild(card);
  });

  // Build psychology items (own storage namespace, own color, no line-through needed but keep consistent)
  psychItems.forEach((text, iIdx) => {
    const id = `psych_${iIdx}`;
    const row = document.createElement("div");
    row.className = "item";
    row.innerHTML = `<input type="checkbox" id="${id}"><label for="${id}">${text}</label>`;
    psychEl.appendChild(row);
  });

  const allCheckboxIds = () => {
    const ids = [];
    sections.forEach(sec => sec.items.forEach((_, i) => ids.push(`${sec.id}_${i}`)));
    return ids;
  };
  const psychCheckboxIds = () => psychItems.map((_, i) => `psych_${i}`);

  // ---- Storage ----
  function loadState(){
    try{
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) : {};
    }catch(e){ return {}; }
  }
  function saveState(state){
    try{ localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); }
    catch(e){ /* storage unavailable, fail silently */ }
  }

  let state = loadState();

  function applyStateToDOM(){
    allCheckboxIds().concat(psychCheckboxIds()).forEach(id => {
      const cb = document.getElementById(id);
      if(!cb) return;
      cb.checked = !!state[id];
      cb.parentElement.classList.toggle("checked", cb.checked);
    });
    const master = document.getElementById("masterCheck");
    master.checked = !!state["master"];
  }

  function updateCounts(){
    sections.forEach(sec => {
      const done = sec.items.filter((_, i) => state[`${sec.id}_${i}`]).length;
      const el = document.querySelector(`[data-count="${sec.id}"]`);
      if(el) el.textContent = `${done}/${sec.items.length}`;
    });
  }

  function updateProgress(){
    const ids = allCheckboxIds();
    const done = ids.filter(id => state[id]).length;
    const total = ids.length;
    const pct = total ? Math.round((done/total)*100) : 0;

    document.getElementById("progressFill").style.width = pct + "%";
    document.getElementById("progressText").textContent = `${done} / ${total} checks complete`;
    document.getElementById("progressPct").textContent = pct + "%";

    const allDone = done === total;
    const masterItem = document.getElementById("masterItem");
    const master = document.getElementById("masterCheck");

    masterItem.classList.toggle("enabled", allDone);
    if(!allDone && master.checked){
      master.checked = false;
      state["master"] = false;
    }

    const light = document.getElementById("light");
    const panelStatus = document.getElementById("panelStatus");
    const panelSub = document.getElementById("panelSub");
    const verdict = document.getElementById("verdict");

    if(state["master"] && allDone){
      light.className = "light go";
      panelStatus.className = "panel-status go";
      panelStatus.textContent = "GO";
      panelSub.textContent = "All checks confirmed. Enter per plan. Do not move the stop.";
      verdict.className = "gate-verdict go";
      verdict.innerHTML = "Enter trade.<span class='small'>Do not move stop loss. Manage according to plan only.</span>";
    } else if(allDone){
      light.className = "light hold";
      panelStatus.className = "panel-status hold";
      panelStatus.textContent = "HOLD";
      panelSub.textContent = "All checks complete — confirm below to unlock entry.";
      verdict.className = "gate-verdict locked";
      verdict.innerHTML = "Confirm the box above to enter.<span class='small'>This is the last gate before the ticket.</span>";
    } else {
      light.className = "light nogo";
      panelStatus.className = "panel-status nogo";
      panelStatus.textContent = "NO-GO";
      panelSub.textContent = done === 0 ? "Checklist not started" : `${total - done} check${total-done===1?"":"s"} remaining`;
      verdict.className = "gate-verdict nogo";
      verdict.innerHTML = "Do not trade.<span class='small'>Finish every section first — no exceptions.</span>";
    }
  }

  function refreshAll(){
    applyStateToDOM();
    updateCounts();
    updateProgress();
  }

  // Event delegation for all checkboxes
  document.body.addEventListener("change", (e) => {
    if(e.target && e.target.type === "checkbox"){
      const id = e.target.id;
      state[id] = e.target.checked;
      e.target.parentElement.classList.toggle("checked", e.target.checked);
      if(id === "masterCheck"){
        state["master"] = e.target.checked;
      }
      saveState(state);
      updateCounts();
      updateProgress();
    }
  });

  document.getElementById("resetBtn").addEventListener("click", () => {
    if(confirm("Reset the checklist for a new trade? This clears every box.")){
      state = {};
      saveState(state);
      refreshAll();
    }
  });

  document.getElementById("printBtn").addEventListener("click", () => window.print());

  refreshAll();
})();
</script>
